cmake_minimum_required(VERSION 3.15)
project(goose_sqlite_native LANGUAGES C)
find_package(SQLite3 3.37 REQUIRED)
find_package(Threads REQUIRED)
add_library(goose_sqlite_native STATIC native/sqlite.c)
set_target_properties(goose_sqlite_native PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(goose_sqlite_native PRIVATE c_std_11)
target_compile_options(goose_sqlite_native PRIVATE -Wall -Wextra -Werror)
if(TARGET SQLite3::SQLite3)
    target_link_libraries(goose_sqlite_native PUBLIC SQLite3::SQLite3 Threads::Threads)
else()
    target_link_libraries(goose_sqlite_native PUBLIC SQLite::SQLite3 Threads::Threads)
endif()
option(GOOSE_SQLITE_NATIVE_TESTS "Build native sanitizer tests" OFF)
option(GOOSE_SQLITE_FUZZ "Build libFuzzer harness" OFF)
if(GOOSE_SQLITE_NATIVE_TESTS OR GOOSE_SQLITE_FUZZ)
    set(sanitizers -O1 -g -fno-omit-frame-pointer -fsanitize=address,undefined)
    target_compile_options(goose_sqlite_native PRIVATE ${sanitizers})
    target_link_options(goose_sqlite_native INTERFACE -fsanitize=address,undefined)
endif()
if(GOOSE_SQLITE_NATIVE_TESTS)
    add_executable(goose_sqlite_test tests/native.c)
    target_compile_options(goose_sqlite_test PRIVATE ${sanitizers} -Wall -Wextra -Werror -UNDEBUG)
    target_link_libraries(goose_sqlite_test PRIVATE goose_sqlite_native)
endif()
if(GOOSE_SQLITE_FUZZ)
    if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang")
        message(FATAL_ERROR "libFuzzer requires Clang")
    endif()
    target_compile_options(goose_sqlite_native PRIVATE -fsanitize=fuzzer-no-link)
    add_executable(goose_sqlite_fuzz tests/fuzz.c)
    target_compile_options(goose_sqlite_fuzz PRIVATE ${sanitizers} -fsanitize=fuzzer)
    target_link_options(goose_sqlite_fuzz PRIVATE -fsanitize=fuzzer)
    target_link_libraries(goose_sqlite_fuzz PRIVATE goose_sqlite_native)
endif()
